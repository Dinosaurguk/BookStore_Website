"""Разовая миграция: заменяет пароли открытым текстом в [Покупатель] на хеши.

Запуск: положи рядом с server.py, закрой Access, сделай копию database.accdb, затем
    python migrate_passwords.py
"""
import os
import sys

import pyodbc
from werkzeug.security import generate_password_hash

BASE_DIR = os.path.dirname(os.path.abspath(sys.argv[0]))
DB_PATH = os.path.join(BASE_DIR, "database.accdb")  # тот же путь, что в server.py
CONN_STR = f"Driver={{Microsoft Access Driver (*.mdb, *.accdb)}};Dbq={DB_PATH};"

HASH_PREFIXES = ("scrypt:", "pbkdf2:")  # так начинаются хеши werkzeug


def main():
    conn = pyodbc.connect(CONN_STR)
    cur = conn.cursor()
    cur.execute("SELECT id_user, Пароль FROM [Покупатель] WHERE Пароль IS NOT NULL")
    rows = cur.fetchall()

    updated = 0
    for id_user, password in rows:
        if not password or password.startswith(HASH_PREFIXES):
            continue  # пустой или уже хеш
        cur.execute(
            "UPDATE [Покупатель] SET Пароль = ? WHERE id_user = ?",
            (generate_password_hash(password), id_user),
        )
        updated += 1

    conn.commit()
    conn.close()
    print(f"Готово. Захешировано паролей: {updated}")


if __name__ == "__main__":
    main()
